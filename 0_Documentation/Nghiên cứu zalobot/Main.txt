var COMMANDS = {};  // Khởi tạo trong _initCommands()

function _initCommands() {
  COMMANDS = {};
  COMMANDS[CONFIG.CMD_LOGIN] = handleLogin;
  COMMANDS[CONFIG.CMD_LOGOUT] = handleLogout;
  COMMANDS[CONFIG.CMD_HELP] = handleHelp;
  COMMANDS[CONFIG.CMD_SEARCH_PHONE] = handleSearch;
  COMMANDS[CONFIG.CMD_SEARCH_TAX] = handleSearch;
  COMMANDS[CONFIG.CMD_SEARCH_NAME] = handleSearch;
  COMMANDS[CONFIG.CMD_SEARCH_CODE] = handleSearch;
  // Thêm lệnh mới ở đây:
  // COMMANDS["/newcmd"] = handleNewCommand;
}

// ============================================================
// WEBHOOK HANDLERS
// ============================================================

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    logToSheet("WEBHOOK", "Nhận request", body);

    var data = body.result || body;
    var eventName = data.event_name;
    var message = data.message;

    if (!message || !message.chat || !message.chat.id) {
      return _buildResponse("No chat info");
    }

    var chatId = message.chat.id;
    var senderName = message.from ? message.from.display_name : "B\u1EA1n";

    switch (eventName) {
      case "message.text.received":
        _routeTextMessage(chatId, senderName, message.text || "");
        break;
      case "message.image.received":
      case "message.sticker.received":
      case "message.unsupported.received":
        sendMessage(chatId, "Hi\u1EC7n FBM Assistant ch\u1EC9 h\u1ED7 tr\u1EE3 tin nh\u1EAFn v\u0103n b\u1EA3n.");
        break;
      default:
        logToSheet("WEBHOOK", "Unknown event: " + eventName);
        break;
    }

    return _buildResponse("OK");
  } catch (error) {
    logToSheet("ERROR", "doPost: " + error.message, error.stack);
    return _buildResponse("Error");
  }
}

function doGet(e) {
  if (e && e.parameter && e.parameter.action === "updateProxy") {
    var newUrl = e.parameter.url;
    if (newUrl) {
      var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
      var sheet = ss.getSheetByName("Config");
      if (!sheet) return ContentService.createTextOutput("ERROR: No Config sheet");

      var data = sheet.getDataRange().getValues();
      for (var i = 0; i < data.length; i++) {
        if (data[i][0] === "PROXY_URL") {
          sheet.getRange(i + 1, 2).setValue(newUrl);
          logToSheet("PROXY", "URL updated", newUrl);
          return ContentService.createTextOutput("OK: " + newUrl);
        }
      }

      sheet.appendRow(["PROXY_URL", newUrl]);
      logToSheet("PROXY", "URL created", newUrl);
      return ContentService.createTextOutput("OK: " + newUrl);
    }
  }

  return ContentService.createTextOutput("FBM Assistant is running");
}


// ============================================================
// ROUTER CHÍNH
// ============================================================

function _routeTextMessage(chatId, senderName, text) {
  sendTyping(chatId);
  _initCommands();

  var trimmed = text.trim();
  if (!trimmed) {
    sendMessage(chatId, "Vui l\u00F2ng nh\u1EADp t\u1EEB kh\u00F3a. G\u00F5 /help \u0111\u1EC3 xem h\u01B0\u1EDBng d\u1EABn.");
    return;
  }

  var lower = trimmed.toLowerCase();

  // 1. Kiểm tra lệnh prefix
  for (var cmd in COMMANDS) {
    if (lower === cmd || lower.indexOf(cmd + " ") === 0) {
      COMMANDS[cmd](chatId, senderName, trimmed);
      return;
    }
  }

  // 2. Kiểm tra session
  var session = getSession(chatId);
  if (!session) {
    _sendLoginPrompt(chatId, senderName);
    return;
  }

  // 3. Tra cứu tự do
  handleFreeSearch(chatId, senderName, trimmed);
}

/** Xử lý /login user|pass */
function handleLogin(chatId, senderName, text) {
  // Tách lấy phần sau "/login "
  var content = text.substring(CONFIG.CMD_LOGIN.length).trim();
  var parts = content.split("|");

  if (parts.length !== 2 || !parts[0].trim() || !parts[1].trim()) {
    sendMessage(chatId,
      "\u274C Sai format. Vui l\u00F2ng nh\u1EADp:\n"
      + CONFIG.CMD_LOGIN + " t\u00EAn\u0111\u0103ngnhap|m\u1EADtkh\u1EA9u\n"
      + "V\u00ED d\u1EE5: " + CONFIG.CMD_LOGIN + " anhlt|matkhau123"
    );
    return;
  }

  var username = parts[0].trim();
  var password = parts[1].trim();

  sendMessage(chatId, "\u23F3 \u0110ang \u0111\u0103ng nh\u1EADp FBM...");

  var loginResult = loginFBM(username, password);
  if (!loginResult) {
    sendMessage(chatId,
      "\u274C Sai th\u00F4ng tin \u0111\u0103ng nh\u1EADp ho\u1EB7c kh\u00F4ng k\u1EBFt n\u1ED1i \u0111\u01B0\u1EE3c FBM.\n"
      + "Vui l\u00F2ng th\u1EED l\u1EA1i: " + CONFIG.CMD_LOGIN + " user|pass"
    );
    return;
  }

  // Mã hóa password
  var now = Date.now();
  var encryptSalt = String(now);
  var encryptedPwd = encryptPassword(password, encryptSalt);

  // Lưu session
  var sessionData = {
    chatId: chatId,
    displayName: senderName,
    fbmUser: username,
    fbmPassword: encryptedPwd,
    encryptSalt: encryptSalt,
    fbmCookie: loginResult.fullCookie,
    database: loginResult.database,
    unit: loginResult.unit,
    userId: loginResult.userId,
    loginTime: now,
    fbmPayloadCookie: loginResult.fbmPayloadCookie,  // Cookie payload cho FBM API search
  };

  // Dùng replaceSession để xóa dòng cũ (nếu có) + thêm mới
  replaceSession(chatId, sessionData);

  sendMessage(chatId,
    "\u2705 \u0110\u0103ng nh\u1EADp th\u00E0nh c\u00F4ng!\n"
    + "User: " + username + "\n"
    + "Database: " + loginResult.database + "\n\n"
    + "B\u1EA1n c\u00F3 th\u1EC3 b\u1EAFt \u0111\u1EA7u tra c\u1EE9u kh\u00E1ch h\u00E0ng.\n"
    + "G\u00F5 /help \u0111\u1EC3 xem h\u01B0\u1EDBng d\u1EABn."
  );
}

}

/** Xử lý /sdt, /mst, /ten, /mkh */
function handleSearch(chatId, senderName, text) {
  var session = getSession(chatId);
  if (!session) {
    _sendLoginPrompt(chatId, senderName);
    return;
  }

  try {
    logToSheet("SEARCH", senderName + ": " + text);
    var result = searchCustomer(chatId, text);
    var messages = formatSearchResult(result, chatId);
    sendMessages(chatId, messages);
    logToSheet("SEARCH_RESULT", "Found: " + result.totalFound + " for: " + result.keyword);
  } catch (error) {
    logToSheet("ERROR", "handleSearch: " + error.message, error.stack);
    sendMessage(chatId, "\u26A0\uFE0F \u0110\u00E3 x\u1EA3y ra l\u1ED7i khi tra c\u1EE9u. Vui l\u00F2ng th\u1EED l\u1EA1i sau.");
  }
}

/** Xử lý text tự do (tra cứu tự động) */
function handleFreeSearch(chatId, senderName, text) {
  try {
    logToSheet("SEARCH", senderName + ": " + text);
    var result = searchCustomer(chatId, text);
    var messages = formatSearchResult(result, chatId);
    sendMessages(chatId, messages);
    logToSheet("SEARCH_RESULT", "Found: " + result.totalFound + " for: " + result.keyword);
  } catch (error) {
    logToSheet("ERROR", "handleFreeSearch: " + error.message, error.stack);
    sendMessage(chatId, "\u26A0\uFE0F \u0110\u00E3 x\u1EA3y ra l\u1ED7i khi tra c\u1EE9u. Vui l\u00F2ng th\u1EED l\u1EA1i sau.");
  }
}

function _buildResponse(message) {
  return ContentService
    .createTextOutput(JSON.stringify({ message: message }))
    .setMimeType(ContentService.MimeType.JSON);
}
